    local function updateFPSCounter()
        local frames = 0
        local lastUpdate = tick()
        
        RunService.Heartbeat:Connect(function()
            frames = frames + 1
            local currentTime = tick()
            
            if currentTime - lastUpdate >= 1 then
                local fps = math.floor(frames / (currentTime - lastUpdate))
                fpsLabel.Text = "FPS: " .. fps
                frames = 0
                lastUpdate = currentTime
            end
        end)

    -- FPS Counter Update Function
    local function updateFPSCounter()
        local frames = 0
        local lastUpdate = tick()
        
        RunService.Heartbeat:Connect(function()
            frames = frames + 1
            local currentTime = tick()
            
            if currentTime - lastUpdate >= 1 then
                local fps = math.floor(frames / (currentTime - lastUpdate))
                fpsLabel.Text = "FPS: " .. fps
                frames = 0
                lastUpdate = currentTime
            end
        end)
    end

    -- Uptime counter
    local startTime = tick()
    local function updateUptime()
        while wait(1) do
            local elapsed = tick() - startTime
            local hours = math.floor(elapsed / 3600)
            local minutes = math.floor((elapsed % 3600) / 60)
            local seconds = math.floor(elapsed % 60)

            if uptimeLabel then
                uptimeLabel.Text = string.format("%02d:%02d:%02d", hours, minutes, seconds)
            end
        end
    end

    -- Auto-scale based on screen size
    local function updateScale()
        local viewportSize = workspace.CurrentCamera.ViewportSize
        local baseWidth = 1920
        local baseHeight = 1080
        local scaleX = viewportSize.X / baseWidth
        local scaleY = viewportSize.Y / baseHeight
        local scale = math.min(scaleX, scaleY, 1.2)

        titleScale.Scale = scale
        rankScale.Scale = scale
        statsScale.Scale = scale
    end

    workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(updateScale)

                sh = nil,

                local workingName = itemName
                
                if workingName:find("Shiny") then
                    parsed.sh = true
                    workingName = workingName:gsub("Shiny ", ""):gsub(" Shiny", "")
                end

                if enchantName and tier then
                    parsed.baseName = enchantName:gsub("^%s+", ""):gsub("%s+$", "")

                    parsed.baseName = itemName

local Formatint = function(int)
    local Suffix = {"", "k", "M", "B", "T", "Qd", "Qn", "Sx", "Sp", "Oc", "No", "De", "UDe", "DDe", "TDe", "QdDe", "QnDe", "SxDe", "SpDe", "OcDe", "NoDe", "Vg", "UVg", "DVg", "TVg", "QdVg", "QnVg", "SxVg", "SpVg", "OcVg", "NoVg", "Tg", "UTg", "DTg", "TTg", "QdTg", "QnTg", "SxTg", "SpTg", "OcTg", "NoTg", "QdAg", "QnAg", "SxAg", "SpAg", "OcAg", "NoAg", "e141", "e144", "e147", "e150", "e153", "e156", "e159", "e162", "e165", "e168", "e171", "e174", "e177", "e180", "e183", "e186", "e189", "e192", "e195", "e198", "e201", "e204", "e207", "e210", "e213", "e216", "e219", "e222", "e225", "e228", "e231", "e234", "e237", "e240", "e243", "e246", "e249", "e252", "e255", "e258", "e261", "e264", "e267", "e270", "e273", "e276", "e279", "e282", "e285", "e288", "e291", "e294", "e297", "e300", "e303"}
    local Index = 1

    if int < 999 then
        return int
    end
    while int >= 1000 and Index < #Suffix do
        int = int / 1000
        Index = Index + 1
    end
    return string.format("%.2f%s", int, Suffix[Index])
end

local GetAsset = function(Id, pt)
    local Asset = require(Library.Directory.Pets)[Id]
    return string.gsub(Asset and (pt == 1 and Asset.goldenThumbnail or Asset.thumbnail) or "14976456685", "rbxassetid://", "")
end

local GetStats = function(Cmds, Class, ItemTable)
    return Cmds.Get({
        Class = { Name = Class },
        IsA = function(InputClass) return InputClass == Class end,
        GetId = function() return ItemTable.id end,
        StackKey = function()
            return game:GetService("HttpService"):JSONEncode({id = ItemTable.id, sh = ItemTable.sh, pt = ItemTable.pt, tn = ItemTable.tn})
        end
    }) or nil
end

local SendWebhook = function(Id, pt, sh)
    -- INCREMENT THE COUNTER
    if string.find(Id, "Huge") then
        getgenv().PhantomXSessionStats.HugesGained = getgenv().PhantomXSessionStats.HugesGained + 1
        print(string.format("[PhantomX] Huge detected! Session total: +%d", getgenv().PhantomXSessionStats.HugesGained))
    elseif string.find(Id, "Titanic") then
        getgenv().PhantomXSessionStats.TitanicsGained = getgenv().PhantomXSessionStats.TitanicsGained + 1
        print(string.format("[PhantomX] Titanic detected! Session total: +%d", getgenv().PhantomXSessionStats.TitanicsGained))
    elseif string.find(Id, "Gargantuan") then
        getgenv().PhantomXSessionStats.GargantuansGained = getgenv().PhantomXSessionStats.GargantuansGained + 1
        print(string.format("[PhantomX] Gargantuan detected! Session total: +%d", getgenv().PhantomXSessionStats.GargantuansGained))
    end
    
    -- Check if webhook URL is configured
    if not getgenv().PhantomXConfig or not getgenv().PhantomXConfig.WebhookURL or getgenv().PhantomXConfig.WebhookURL == "" then
        return
    end
    
    local Img = string.format("https://biggamesapi.io/image/%s", GetAsset(Id, pt))
    local Version = pt == 1 and "Golden " or pt == 2 and "Rainbow " or ""
    local PetType = string.find(Id, "Huge") and "Huge Pet" or string.find(Id, "Titanic") and "Titanic Pet" or string.find(Id, "Gargantuan") and "Gargantuan Pet" or "Pet"
    local Exist = GetStats(ExistCmds, "Pet", { id = Id, pt = pt, sh = sh, tn = nil })

    local Body = game:GetService("HttpService"):JSONEncode({
        content = (getgenv().PhantomXConfig.DiscordID and getgenv().PhantomXConfig.DiscordID ~= "") and string.format("<@%s>", getgenv().PhantomXConfig.DiscordID) or "",
        embeds = {
            {
                title = string.format("Just got a %s!", PetType),
                color = 0xADD8E6,
                timestamp = DateTime.now():ToIsoDate(),
                thumbnail = { url = Img },
                description = string.format("**%s Info💫:**\n**Name:** %s%s%s\n**Exist count:** `%s\n**In Account:** ||%s||",
                    PetType, Version, sh and "Shiny " or "", Id, Formatint(Exist or 0), LocalPlayer.Name),
                footer = { text = "PhantomX Webhook System" }
            }
        }
    })

    pcall(function()
        local requestFunc = request or http_request or syn.request or HttpPost or http.request
        if requestFunc then
            return requestFunc({
                Url = getgenv().PhantomXConfig.WebhookURL,
                Method = "POST",
                Headers = { ["Content-Type"] = "application/json" },
                Body = Body
            })
        end
    end)
end

-- Store all existing huge/titanic/gargantuan UIDs at startup
for uid, petData in pairs(SaveMod.Get()['Inventory']['Pet'] or {}) do
    if (string.find(petData.id, "Huge") or string.find(petData.id, "Titanic") or string.find(petData.id, "Gargantuan")) then
        StoredUIDs[uid] = true
    end
end

-- Listen for inventory updates
Network.Fired("Items: Update"):Connect(function(_, Inventory)
    if Inventory["set"] and Inventory["set"]["Pet"] then
        for uid, petData in pairs(Inventory["set"]["Pet"]) do
            if (string.find(petData.id, "Huge") or string.find(petData.id, "Titanic") or string.find(petData.id, "Gargantuan")) and not StoredUIDs[uid] then
                SendWebhook(petData.id, petData.pt, petData.sh)
                StoredUIDs[uid] = true
            end
        end
    end
end)
